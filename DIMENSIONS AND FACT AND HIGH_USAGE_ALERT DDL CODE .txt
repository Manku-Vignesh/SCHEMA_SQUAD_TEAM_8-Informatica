1. DIM_CUSTOMER_TEL

CREATE TABLE DIM_CUSTOMER_TEL (
    CUSTOMER_KEY      NUMBER(10),
    PHONE_NUMBER      VARCHAR2(15),
    CUSTOMER_NAME     VARCHAR2(100),
    PLAN_TYPE         VARCHAR2(20)
);

2.DIM_TOWER_TEL

CREATE TABLE DIM_TOWER_TEL (
    TOWER_KEY         NUMBER(10),
    TOWER_ID          VARCHAR2(10),
    REGION            VARCHAR2(50),
    CITY              VARCHAR2(50)
);



3. FACT_CALLS_TEL

CREATE TABLE FACT_CALLS_TEL (
    FACT_ID             NUMBER(10),
    CALL_ID             NUMBER(10),
    CUSTOMER_KEY        NUMBER(10),
    TOWER_KEY           NUMBER(10),
    DATE_KEY            NUMBER(10),
    CALLER_NUMBER       VARCHAR2(15),
    RECEIVER_NUMBER     VARCHAR2(15),
    CALL_DURATION       NUMBER(10,2),
    CALL_TYPE           VARCHAR2(20),
    REVENUE             NUMBER(12,2),
    IS_INTERNATIONAL    NUMBER(1)
);


4.HIGH_USAGE_ALERT

CREATE TABLE HIGH_USAGE_ALERT (
    Alert_ID INT PRIMARY KEY,
    CallID VARCHAR(20),
    Phone_Number VARCHAR(15),
    Duration INT,
    Revenue DECIMAL(10,2),
    Alert_Time TIMESTAMP
);
