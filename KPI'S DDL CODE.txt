CREATE TABLE KPI_1 (
    DATE_KEY INT,
    TOTAL_CALLS INT,
    UNIQUE_CUSTOMERS INT,
    AVG_CALLS DECIMAL(10,2),
    PEAK_DURATION INT
);


CREATE TABLE KPI_2(
CALLTYPE VARCHAR2(50),
TOTAL_CALLS NUMBER,
TOTAL_DURATION NUMBER,
TOTAL_REVENUE NUMBER(10,2),
AVG_REVENUE_PER_CALL NUMBER(10,2)
);

CREATE TABLE KPI_3 (
    DATE_KEY NUMBER,
    TOTAL_INT_REVENUE NUMBER(10,2),
    TOTAL_INT_CALLS NUMBER,
    INT_REVENUE_PER_CALL NUMBER(10,2)
)


CREATE TABLE KPI_4(
CUSTOMER_ID NUMBER,
TOTAL_REVENUE NUMBER(10,2),
PERCENT_CONTRIBUTION NUMBER(10,2),
RANK_NO NUMBER
);
